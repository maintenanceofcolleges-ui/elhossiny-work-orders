/** * فحص الصفوف الفاضية والمكررة في شيت أوامر العمل. * مبيمسحش حاجة: بيلوّن الصفوف المشكوك فيها بالأصفر * وبيعمل شيت جديد اسمه "فحص_التكرار" فيه أرقام الصفوف. * شغّله مرة واحدة من القائمة: أدوات الفحص ← فحص الصفوف. */ const SHEET_NAME = 'Sheet1'; // غيّره لو اسم شيت البيانات مختلف (اسمه ظاهر تحت في التبويب) function onOpen() { SpreadsheetApp.getUi().createMenu('أدوات الفحص') .addItem('فحص الصفوف', 'checkRows') .addItem('إزالة التلوين', 'clearHighlight') .addToUi(); } function checkRows() { const ss = SpreadsheetApp.getActive(); const sh = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0]; const lastRow = sh.getLastRow(), lastCol = sh.getLastColumn(); const values = sh.getRange(1, 1, lastRow, lastCol).getValues(); const seen = {}; const blank = [], dup = []; for (let i = 1; i < values.length; i++) { // تخطي الهيدر const row = values[i]; const nonEmpty = row.filter(v => String(v).trim() !== '').length; if (nonEmpty <= 1) { blank.push(i + 1); continue; } // صف فاضي أو فيه رقم بس // مفتاح التكرار: كل الخلايا ما عدا أول عمودين (تاريخ/رقم الأمر) const key = row.slice(2).map(v => String(v).trim()).join('|'); if (seen[key]) dup.push({ row: i + 1, sameAs: seen[key] }); else seen[key] = i + 1; } // تلوين [...blank, ...dup.map(d => d.row)].forEach(r => sh.getRange(r, 1, 1, lastCol).setBackground('#FFF2CC')); // تقرير let log = ss.getSheetByName('فحص_التكرار'); if (log) log.clear(); else log = ss.insertSheet('فحص_التكرار'); const out = [['النوع', 'رقم الصف', 'مطابق للصف']]; blank.forEach(r => out.push(['صف فاضي', r, ''])); dup.forEach(d => out.push(['مكرر', d.row, d.sameAs])); log.getRange(1, 1, out.length, 3).setValues(out); SpreadsheetApp.getUi().alert( 'صفوف فاضية: ' + blank.length + '\nصفوف مكررة: ' + dup.length + '\n\nاتلونت بالأصفر، وتفاصيلها في شيت "فحص_التكرار".'); } function clearHighlight() { const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActive().getSheets()[0]; sh.getRange(2, 1, sh.getLastRow() - 1, sh.getLastColumn()).setBackground(null); }
